cmake_minimum_required(VERSION 3.20)
project(kukri VERSION 0.1.0 LANGUAGES CXX)
option(KUKRI_REQUIRE_TEST_DEPENDENCIES "Fail when integration-test dependencies are missing" OFF)
option(KUKRI_STATIC_MSVC_RUNTIME "Link the MSVC runtime statically for portable packages" ON)
if(MSVC AND KUKRI_STATIC_MSVC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
configure_file(cmake/version.hpp.in generated/kukri/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(kukri_core src/cleaner.cpp src/language.cpp src/process.cpp src/attributes.cpp src/git.cpp)
target_include_directories(kukri_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated" include)
if(MSVC)
  target_compile_options(kukri_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(kukri_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(kukri src/main.cpp)
target_link_libraries(kukri PRIVATE kukri_core)
# Relative destinations support --prefix and package-manager staging (DESTDIR).
include(GNUInstallDirs)
install(TARGETS kukri RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# Keep installations independent of the ignored local learning directory.
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(MINGW)
  target_link_options(kukri PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit.cpp)
  target_link_libraries(unit_tests PRIVATE kukri_core)
  add_test(NAME unit COMMAND unit_tests)
  add_executable(language_tests tests/languages.cpp)
  target_link_libraries(language_tests PRIVATE kukri_core)
  add_test(NAME languages COMMAND language_tests)
  if(KUKRI_REQUIRE_TEST_DEPENDENCIES)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    find_package(Git 2.31 REQUIRED)
    find_program(KUKRI_NODE_EXECUTABLE NAMES node REQUIRED)
  else()
    find_package(Python3 3.9 COMPONENTS Interpreter)
    find_package(Git 2.31)
  endif()
  if(Python3_FOUND)
    add_test(NAME language_semantics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/language_semantics.py $<TARGET_FILE:kukri>)
  endif()
  if(Python3_FOUND AND GIT_FOUND)
    add_test(NAME git_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:kukri> ${GIT_EXECUTABLE} ${CMAKE_COMMAND} ${CMAKE_BINARY_DIR} $<CONFIG>)
  else()
    message(WARNING "Python 3 and Git are needed for integration tests")
  endif()
endif()

include(cmake/Packaging.cmake)
